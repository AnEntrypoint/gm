function(gmnew_host_link_worker)
    get_property(host_dir GLOBAL PROPERTY GMNEW_HOST_DIR)
    set(missing)
    foreach(needed IN ITEMS gmnew_mcp gmnew_verbs gmnew_search gmnew_runcode gmnew_mcpclient)
        if(NOT TARGET ${needed})
            list(APPEND missing ${needed})
        endif()
    endforeach()
    set(leaf_targets)
    if(TARGET gmnew_leaves)
        list(APPEND leaf_targets gmnew_leaves)
    endif()
    if(missing)
        list(JOIN missing ", " missing_text)
        message(STATUS "gmnew: worker and doctor modes wait for ${missing_text}")
        target_sources(gmnew PRIVATE ${host_dir}/worker_unavailable.cpp)
        target_compile_definitions(gmnew PRIVATE GMNEW_HOST_MISSING_TARGETS="${missing_text}")
        return()
    endif()
    target_sources(gmnew PRIVATE ${host_dir}/worker.cpp)
    target_link_libraries(gmnew PRIVATE gmnew_mcp gmnew_verbs gmnew_search gmnew_runcode gmnew_mcpclient ${leaf_targets})
    if(GMNEW_BUILD_TESTS)
        add_executable(gmnew_worker_test ${host_dir}/tests/worker_test.cpp)
        target_link_libraries(gmnew_worker_test PRIVATE gmnew_host gmnew_runcode)
        target_compile_definitions(gmnew_worker_test PRIVATE GMNEW_BINARY="$<TARGET_FILE:gmnew>"
                                                             GMNEW_SOURCE_ROOT="${PROJECT_SOURCE_DIR}")
        add_dependencies(gmnew_worker_test gmnew)
        add_test(NAME gmnew_worker_test COMMAND gmnew_worker_test)
    endif()
endfunction()
set_property(GLOBAL PROPERTY GMNEW_HOST_DIR ${CMAKE_CURRENT_SOURCE_DIR})
cmake_language(DEFER DIRECTORY ${PROJECT_SOURCE_DIR} CALL gmnew_host_link_worker)
