function(gm_host_link_worker)
    get_property(host_dir GLOBAL PROPERTY GMNEW_HOST_DIR)
    set(missing)
    foreach(needed IN ITEMS gm_mcp gm_verbs gm_search gm_runcode gm_mcpclient)
        if(NOT TARGET ${needed})
            list(APPEND missing ${needed})
        endif()
    endforeach()
    set(leaf_targets)
    if(TARGET gm_leaves)
        list(APPEND leaf_targets gm_leaves)
    endif()
    if(missing)
        list(JOIN missing ", " missing_text)
        message(STATUS "gm: worker and doctor modes wait for ${missing_text}")
        target_sources(gm PRIVATE ${host_dir}/worker_unavailable.cpp)
        target_compile_definitions(gm PRIVATE GMNEW_HOST_MISSING_TARGETS="${missing_text}")
        return()
    endif()
    target_sources(gm PRIVATE ${host_dir}/worker.cpp)
    target_link_libraries(gm PRIVATE gm_mcp gm_verbs gm_search gm_runcode gm_mcpclient ${leaf_targets})
    if(GMNEW_BUILD_TESTS)
        add_executable(gm_worker_test ${host_dir}/tests/worker_test.cpp)
        target_link_libraries(gm_worker_test PRIVATE gm_host gm_runcode)
        target_compile_definitions(gm_worker_test PRIVATE GMNEW_BINARY="$<TARGET_FILE:gm>"
                                                             GMNEW_SOURCE_ROOT="${PROJECT_SOURCE_DIR}")
        add_dependencies(gm_worker_test gm)
        add_test(NAME gm_worker_test COMMAND gm_worker_test)
    endif()
endfunction()
set_property(GLOBAL PROPERTY GMNEW_HOST_DIR ${CMAKE_CURRENT_SOURCE_DIR})
cmake_language(DEFER DIRECTORY ${PROJECT_SOURCE_DIR} CALL gm_host_link_worker)
