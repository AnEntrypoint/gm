file(READ ${CMAKE_CURRENT_LIST_DIR}/../../../package.json GMNEW_MCP_PACKAGE_JSON)
string(JSON GMNEW_MCP_PACKAGE_VERSION GET ${GMNEW_MCP_PACKAGE_JSON} version)

add_library(gm_mcp STATIC ${CMAKE_CURRENT_LIST_DIR}/server.cpp ${CMAKE_CURRENT_LIST_DIR}/skills.cpp ${CMAKE_CURRENT_LIST_DIR}/sha256.cpp ${CMAKE_CURRENT_LIST_DIR}/entry.cpp ${CMAKE_CURRENT_LIST_DIR}/compact.cpp ${CMAKE_CURRENT_LIST_DIR}/pending.cpp)
target_include_directories(gm_mcp PUBLIC ${CMAKE_CURRENT_LIST_DIR}/include ${CMAKE_CURRENT_LIST_DIR}/../../include)
target_compile_definitions(gm_mcp PUBLIC GMNEW_MCP_VERSION="${GMNEW_MCP_PACKAGE_VERSION}")
find_package(Threads REQUIRED)
target_link_libraries(gm_mcp PUBLIC gm_json gm_log gm_proc Threads::Threads $<TARGET_NAME_IF_EXISTS:gm_runcode>)

if(GMNEW_BUILD_TESTS)
    add_executable(gm_mcp_test ${CMAKE_CURRENT_LIST_DIR}/tests/mcp_test.cpp)
    target_link_libraries(gm_mcp_test PRIVATE gm_mcp)
    target_compile_definitions(gm_mcp_test PRIVATE GMNEW_SOURCE_DIR="${CMAKE_CURRENT_LIST_DIR}/../../.."
                               $<$<TARGET_EXISTS:gm_runcode>:GMNEW_HAVE_RUNCODE>)
    add_test(NAME gm_mcp_test COMMAND gm_mcp_test)
    add_executable(gm_mcp_spec_test ${CMAKE_CURRENT_LIST_DIR}/tests/spec_test.cpp)
    target_link_libraries(gm_mcp_spec_test PRIVATE gm_mcp)
    target_compile_definitions(gm_mcp_spec_test PRIVATE GMNEW_SOURCE_DIR="${CMAKE_CURRENT_LIST_DIR}/../../..")
    foreach(section discover caching missing annotations capabilities list_changed)
        add_test(NAME gm_mcp_spec_${section} COMMAND gm_mcp_spec_test ${section})
    endforeach()
    add_executable(gm_mcp_schema_test ${CMAKE_CURRENT_LIST_DIR}/tests/schema_test.cpp)
    target_link_libraries(gm_mcp_schema_test PRIVATE gm_mcp)
    target_compile_definitions(gm_mcp_schema_test PRIVATE GMNEW_SOURCE_DIR="${CMAKE_CURRENT_LIST_DIR}/../../..")
    add_test(NAME gm_mcp_schema COMMAND gm_mcp_schema_test)
endif()
